** Binds
#+begin_src lua :tangle ~/.config/hypr/hyprland.lua
  -- Layout messages
  hl.bind("SUPER + COMMA",  hl.dsp.layout("addmaster"))
  hl.bind("SUPER + PERIOD", hl.dsp.layout("removemaster"))

  -- Launch applications
  hl.bind("SUPER + Return",           hl.dsp.exec_cmd(C_TERMINAL))
  hl.bind("SUPER + I",                hl.dsp.exec_cmd(C_BROWSER))
  hl.bind("SUPER + SHIFT + I",        hl.dsp.exec_cmd(C_BROWSER_PRIVATE))
  hl.bind("SUPER + E",                hl.dsp.exec_cmd(C_EMACS))
  hl.bind("SUPER + R",                hl.dsp.exec_cmd(C_ROFI))
  hl.bind("SUPER + SHIFT + CTRL + Q", hl.dsp.exec_cmd(C_POWEROFF))
  hl.bind("SUPER + SHIFT + CTRL + R", hl.dsp.exec_cmd(C_REBOOT))
  hl.bind("XF86AudioRaiseVolume",     hl.dsp.exec_cmd(C_RAISE_VOLUME))
  hl.bind("XF86AudioLowerVolume",     hl.dsp.exec_cmd(C_LOWER_VOLUME))
  hl.bind("SUPER + A",                hl.dsp.exec_cmd(C_SWITCH_SINK))
  hl.bind("SUPER + C",                hl.dsp.exec_cmd(C_CHANGE_COLOR))
  hl.bind("XF86AudioMute",            hl.dsp.exec_cmd(C_MUTE_AUDIO))
  hl.bind("XF86AudioPlay",            hl.dsp.exec_cmd(C_PLAY_PAUSE_AUDIO))
  hl.bind("XF86AudioStop",            hl.dsp.exec_cmd(C_STOP_AUDIO))
  hl.bind("XF86AudioNext",            hl.dsp.exec_cmd(C_NEXT_AUDIO))
  hl.bind("XF86AudioPrev",            hl.dsp.exec_cmd(C_PREV_AUDIO))
  hl.bind("XF86AudioMicMute",         hl.dsp.exec_cmd(C_MUTE_MIC))
  hl.bind("XF86MonBrightnessUp",      hl.dsp.exec_cmd(C_RAISE_BRIGHTNESS))
  hl.bind("XF86MonBrightnessDown",    hl.dsp.exec_cmd(C_LOWER_BRIGHTNESS))
  hl.bind("XF86Calculator",           hl.dsp.exec_cmd(C_CALCULATOR))
  hl.bind("Print",                    hl.dsp.exec_cmd(C_SCREENSHOT))
  hl.bind("SHIFT + Print",            hl.dsp.exec_cmd(C_SCREENSHOT_EDIT))
  hl.bind("SUPER + SHIFT + W",        hl.dsp.exec_cmd(C_KILL_WINDOW))

  -- Scratch terminal
  hl.bind("SUPER + SHIFT + Return", function ()
             if hl.get_window("class:" .. SCRATCH_CLASS) == nil then
                hl.dispatch(hl.dsp.exec_cmd("alacritty --class " .. SCRATCH_CLASS))
             else
                hl.dispatch(hl.dsp.workspace.toggle_special("scratch"))
             end
  end)

  -- Mouse move / resize
  hl.bind("SUPER + mouse:272", hl.dsp.window.drag(),   { mouse = true })
  hl.bind("SUPER + mouse:273", hl.dsp.window.resize(), { mouse = true })

  -- Window management
  hl.bind("SUPER + CTRL + Q", hl.dsp.exit())
  hl.bind("SUPER + W",        hl.dsp.window.close())
  hl.bind("SUPER + F",        hl.dsp.window.fullscreen())
  hl.bind("SUPER + space",    hl.dsp.window.float())
  hl.bind("SUPER + P",        function ()
             hl.dispatch(hl.dsp.window.float({ action = "enable" }))
             hl.dispatch(hl.dsp.window.pin())
  end)

  -- Focus direction
  hl.bind("SUPER + H",     hl.dsp.focus({ direction = "left" }))
  hl.bind("SUPER + LEFT",  hl.dsp.focus({ direction = "left" }))
  hl.bind("SUPER + J",     hl.dsp.focus({ direction = "down" }))
  hl.bind("SUPER + DOWN",  hl.dsp.focus({ direction = "down" }))
  hl.bind("SUPER + K",     hl.dsp.focus({ direction = "up" }))
  hl.bind("SUPER + UP",    hl.dsp.focus({ direction = "up" }))
  hl.bind("SUPER + L",     hl.dsp.focus({ direction = "right" }))
  hl.bind("SUPER + RIGHT", hl.dsp.focus({ direction = "right" }))

  -- Move window
  hl.bind("SUPER + SHIFT + H",     hl.dsp.window.move({ direction = "left" }))
  hl.bind("SUPER + SHIFT + LEFT",  hl.dsp.window.move({ direction = "left" }))
  hl.bind("SUPER + SHIFT + J",     hl.dsp.window.move({ direction = "down" }))
  hl.bind("SUPER + SHIFT + DOWN",  hl.dsp.window.move({ direction = "down" }))
  hl.bind("SUPER + SHIFT + K",     hl.dsp.window.move({ direction = "up" }))
  hl.bind("SUPER + SHIFT + UP",    hl.dsp.window.move({ direction = "up" }))
  hl.bind("SUPER + SHIFT + L",     hl.dsp.window.move({ direction = "right" }))
  hl.bind("SUPER + SHIFT + RIGHT", hl.dsp.window.move({ direction = "right" }))

  -- Resize window
  hl.bind("SUPER + CTRL + H",     hl.dsp.window.resize({ x = -80, y = 0,   relative = true }))
  hl.bind("SUPER + CTRL + LEFT",  hl.dsp.window.resize({ x = -80, y = 0,   relative = true }))
  hl.bind("SUPER + CTRL + J",     hl.dsp.window.resize({ x = 0,   y = 80,  relative = true }))
  hl.bind("SUPER + CTRL + DOWN",  hl.dsp.window.resize({ x = 0,   y = 80,  relative = true }))
  hl.bind("SUPER + CTRL + K",     hl.dsp.window.resize({ x = 0,   y = -80, relative = true }))
  hl.bind("SUPER + CTRL + UP",    hl.dsp.window.resize({ x = 0,   y = -80, relative = true }))
  hl.bind("SUPER + CTRL + L",     hl.dsp.window.resize({ x = 80,  y = 0,   relative = true }))
  hl.bind("SUPER + CTRL + RIGHT", hl.dsp.window.resize({ x = 80,  y = 0,   relative = true }))

  -- Workspaces (loop for 1–9, then explicit 10)
  for i = 1, 9 do
     hl.bind("SUPER + " .. i,         hl.dsp.focus({ workspace = i }))
     hl.bind("SUPER + SHIFT + " .. i, hl.dsp.window.move({ workspace = i, follow = false }))
  end
  hl.bind("SUPER + 0",         hl.dsp.focus({ workspace = 10 }))
  hl.bind("SUPER + SHIFT + 0", hl.dsp.window.move({ workspace = 10, follow = false }))

  hl.bind("SUPER + TAB",        hl.dsp.focus({ workspace = "previous" }))
  hl.bind("SUPER + mouse_down", hl.dsp.focus({ workspace = "e-1" }))
  hl.bind("SUPER + mouse_up",   hl.dsp.focus({ workspace = "e+1" }))
#+end_src
